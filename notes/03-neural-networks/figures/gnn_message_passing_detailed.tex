\begin{tikzpicture}[

    \draw[edge] (i1) -- (k1);
    \draw[edge] (i1) -- (l1);
    \draw[edge] (j1) -- (k1);
\end{scope}

% Arrow to stage 2

% === STAGE 2: COMPUTE MESSAGES ===
\node[stage] at (-0.3, 2) {2. Compute messages};

\begin{scope}[shift={(-0.3, 0)}]
    \node[target] (i2) at (0, 0) {$h_i$};
    \node[neighbor] (j2) at (-1.2, 0.8) {$h_j$};
